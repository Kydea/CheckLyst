PROTOTIO DI TRIGGER:

create function "CheckLists"."TF_Versions_Validate"() returns trigger as 
$body$
declare
	_count int = -1;
begin
	select
		_count = count(*)
	from
		"CheckLists"."PrototypeVersions" v
	where
		v."Prototype" = new."Prototype"
		and v."VersionState" = 'Draft'

	if _counter > 1 then 
		raise exception 'A draft version of the prototype % already exists', new."Prototype";
	end if;

	return new;  
end;
$body$ language plpgsql;

create trigger "T_Versions_Validate" 
	after insert or update
	on "CheckLists"."PrototypeVersions" 
	for each row execute procedure "TF_Versions_Validate"();
